-- OLAP01. ROLL-UP
-- Cau hoi: Moi mien co bao nhieu tin dang dat?
-- Du kien tu Cube Browser: Mien Bac 26749; Mien Nam 55152; Mien Trung 27008.
-- Tong 3 mien = 108909.
SELECT
    {[Measures].[Fact Listing Count]} ON COLUMNS,
    NON EMPTY
        [Dim Location].[Region Group].[Region Group].MEMBERS ON ROWS
FROM [Cube_Land];


-- OLAP02. DRILL-DOWN
-- Cau hoi: Khi tu mien di xuong tinh/thanh, so tin phan bo nhu the nao?
-- DrilldownLevel hien dong tong moi mien va cac tinh/thanh ngay ben duoi mien do.
-- LUU Y: KHONG cong tong tat ca cac dong hien thi vi dong mien va dong tinh la 2 cap.
SELECT
    {[Measures].[Fact Listing Count]} ON COLUMNS,
    NON EMPTY
        DrilldownLevel(
            [Dim Location].[Geography].[Region Group].MEMBERS
        ) ON ROWS
FROM [Cube_Land];


-- OLAP03. SLICE
-- Cau hoi: Rieng Mien Bac, so tin theo tung phan khuc gia la bao nhieu?
-- Co dinh 1 mien qua WHERE, giu phan khuc gia lam goc nhin.
-- Tong cac phan khuc du kien = 26749.
SELECT
    {[Measures].[Fact Listing Count]} ON COLUMNS,
    NON EMPTY
        [Dim Price Segment].[Price Segment].[Price Segment].MEMBERS ON ROWS
FROM [Cube_Land]
WHERE ([Dim Location].[Region Group].&[Miền Bắc]);


-- OLAP04. DICE
-- Cau hoi: Tai Mien Nam, trong 2 phan khuc gia 1-3 / 3-5 ty,
-- va 2 nhom dien tich 50-100 / 100-200 m2, so tin phan bo ra sao?
-- Gioi han dong thoi nhieu chieu: khu vuc, phan khuc gia, dien tich.
SELECT
    NON EMPTY {
        [Dim Price Segment].[Price Segment].&[1 - 3 tỷ],
        [Dim Price Segment].[Price Segment].&[3 - 5 tỷ]
    } ON COLUMNS,
    NON EMPTY {
        [Dim Size Group].[Area Group].&[50 - 100 m2],
        [Dim Size Group].[Area Group].&[100 - 200 m2]
    } ON ROWS
FROM [Cube_Land]
WHERE (
    [Measures].[Fact Listing Count],
    [Dim Location].[Region Group].&[Miền Nam]
);


-- OLAP05. PIVOT
-- Cau hoi: Doi hang va cot cua bang gia x dien tich toan quoc co doi so lieu khong?

-- BR01. TIM KHU VUC CO NHIEU TIN DAT PHU HOP NGAN SACH/DIEN TICH
-- Cau hoi: Neu nguoi mua can dat GIA DUOI 3 TY va DIEN TICH TU 100 DEN DUOI
--          200 M2, TOP 10 tinh/thanh nao co nhieu tin nhat? Ty trong quoc gia?
-- Du lieu: Dung DUNG 2 Price Segment "Duoi 1 ty", "1 - 3 ty" va
--          1 Area Group "100 - 200 m2" trong Cube hien tai.
-- Ranh gioi nhom ETL: [0,1), [1,3) ty va [100,200) m2.
-- LUU Y: Gia duoi 3 ty KHAC gia <= 3 ty; gia tri dung 3 ty khong nam trong tap.
--          Neu can 2,7 ty hoac 80-150 m2 thi Dashboard loc gia/area tung tin.
--
-- BR01.1. DOI SOAT TOAN QUOC
-- Du kien tu bang BR03 da kiem thu: 7076 + 11205 = 18281 tin.
-- Chay rieng ca khoi WITH ... SELECT ... ;
WITH
    MEMBER [Measures].[BR01 Matching Listings] AS
        SUM(
            {
                [Dim Price Segment].[Price Segment].&[Dưới 1 tỷ],
                [Dim Price Segment].[Price Segment].&[1 - 3 tỷ]
            },
            (
                [Measures].[Fact Listing Count],
                [Dim Size Group].[Area Group].&[100 - 200 m2]
            )
        ),
        FORMAT_STRING = '#,##0'
SELECT
    {[Measures].[BR01 Matching Listings]} ON COLUMNS
FROM [Cube_Land];


-- BR01.2. TOP 10 TINH/THANH VA TY TRONG SO TIN PHU HOP TOAN QUOC
-- Tong du kien de lam mau so: 18281 (BR01.1). Top 10 chi la mot phan tong.
-- [BR01 Matching Listings] la dem tin, khong la tong gia/dien tich.
WITH
    MEMBER [Measures].[BR01 Matching Listings] AS
        SUM(
            {
                [Dim Price Segment].[Price Segment].&[Dưới 1 tỷ],
                [Dim Price Segment].[Price Segment].&[1 - 3 tỷ]
            },
            (
                [Measures].[Fact Listing Count],
                [Dim Size Group].[Area Group].&[100 - 200 m2]
            )
        ),
        FORMAT_STRING = '#,##0'

    MEMBER [Measures].[BR01 National Matching Listings] AS
        SUM(
            [Dim Location].[Province New].[Province New].MEMBERS,
            [Measures].[BR01 Matching Listings]
        ),
        FORMAT_STRING = '#,##0'

    MEMBER [Measures].[BR01 Share National] AS
        IIF(
            [Measures].[BR01 National Matching Listings] = 0,
            NULL,
            [Measures].[BR01 Matching Listings] /
                [Measures].[BR01 National Matching Listings]
        ),
        FORMAT_STRING = '0.00%'

    SET [BR01 Top 10 Provinces] AS
        TOPCOUNT(
            FILTER(
                [Dim Location].[Province New].[Province New].MEMBERS,
                [Measures].[BR01 Matching Listings] > 0
            ),
            10,
            [Measures].[BR01 Matching Listings]
        )
SELECT
    {
        [Measures].[BR01 Matching Listings],
        [Measures].[BR01 Share National]
    } ON COLUMNS,
    NON EMPTY [BR01 Top 10 Provinces] ON ROWS
FROM [Cube_Land];

-- BR03.5. PARETO 20% TO HOP (6 / 30 TO HOP)
-- Cau hoi: 6 to hop nhom gia x nhom dien tich co nhieu tin nhat
--          chiem bao nhieu % cua 108909 tin toan quoc?
-- Day la phep KIEM TRA Pareto 80/20, khong mac dinh dat 80%.
-- Doi soat tu CSV sach: top 6 = 47303 tin, chiem khoang 43.43%.
-- Cac dong duoc sap xep giam dan theo so tin.
WITH
    SET [BR03 Top 6 Combinations] AS
        TOPCOUNT(
            NONEMPTY(
                CROSSJOIN(
                    [Dim Price Segment].[Price Segment].[Price Segment].MEMBERS,
                    [Dim Size Group].[Area Group].[Area Group].MEMBERS
                ),
                [Measures].[Fact Listing Count]
            ),
            6,
            [Measures].[Fact Listing Count]
        )

    MEMBER [Measures].[BR03 Share National] AS
        IIF(
            ([Measures].[Fact Listing Count],
             [Dim Price Segment].[Price Segment].DefaultMember,
             [Dim Size Group].[Area Group].DefaultMember) = 0,
            NULL,
            [Measures].[Fact Listing Count] /
            ([Measures].[Fact Listing Count],
             [Dim Price Segment].[Price Segment].DefaultMember,
             [Dim Size Group].[Area Group].DefaultMember)
        ),
        FORMAT_STRING = '0.00%'
SELECT
    {
        [Measures].[Fact Listing Count],
        [Measures].[BR03 Share National]
    } ON COLUMNS,
    NON EMPTY [BR03 Top 6 Combinations] ON ROWS
FROM [Cube_Land];


-- BR03.6. PARETO: NHOM NHO NHAT DAT IT NHAT 80% TONG SO TIN
-- TopSum tu tim so to hop it nhat sao cho tong so tin >= 80% toan quoc.
-- Doi soat tu CSV sach: can 16/30 to hop (53.33%),
-- tong 88440 tin (khoang 81.21%); KHONG phai 20% to hop.
-- Neu ket qua khac, kiem tra nhom gia/dien tich da Process trong Cube.
WITH
    SET [BR03 All Combinations] AS
        NONEMPTY(
            CROSSJOIN(
                [Dim Price Segment].[Price Segment].[Price Segment].MEMBERS,
                [Dim Size Group].[Area Group].[Area Group].MEMBERS
            ),
            [Measures].[Fact Listing Count]
        )

    SET [BR03 Pareto 80 Combinations] AS
        TOPSUM(
            [BR03 All Combinations],
            0.8 *
            ([Measures].[Fact Listing Count],
             [Dim Price Segment].[Price Segment].DefaultMember,
             [Dim Size Group].[Area Group].DefaultMember),
            [Measures].[Fact Listing Count]
        )

    MEMBER [Measures].[BR03 Share National] AS
        IIF(
            ([Measures].[Fact Listing Count],
             [Dim Price Segment].[Price Segment].DefaultMember,
             [Dim Size Group].[Area Group].DefaultMember) = 0,
            NULL,
            [Measures].[Fact Listing Count] /
            ([Measures].[Fact Listing Count],
             [Dim Price Segment].[Price Segment].DefaultMember,
             [Dim Size Group].[Area Group].DefaultMember)
        ),
        FORMAT_STRING = '0.00%'
SELECT
    {
        [Measures].[Fact Listing Count],
        [Measures].[BR03 Share National]
    } ON COLUMNS,
    NON EMPTY [BR03 Pareto 80 Combinations] ON ROWS
FROM [Cube_Land];

-- BR04.1. CO CAU SO TIN THEO VI TRI
-- Cau hoi bo tro: Duong chinh, Trong hem, Khong xac dinh co bao nhieu tin?

            ([Measures].[Fact Listing Count],
             [Dim Land Profile].[Position].DefaultMember)

        FORMAT_STRING = '0.00%'

    {
        [Measures].[Fact Listing Count],
        [Measures].[BR04 Share National Listings]
    } ON COLUMNS,
    NON EMPTY
        [Dim Land Profile].[Position].[Position].MEMBERS ON ROWS
FROM [Cube_Land];


-- BR04.2. SO TIN THEO VI TRI x MIEN (CHI DUNG DE DOI CHIEU MAU)
